using System;
using System.Threading;
using Pretzel.Data;
using UnityEngine;

// Generated alongside the game keys. No keys are stored in this script.
// Add to a GameObject; use its context menu in Play mode. Nothing runs automatically.
// Re-exporting keys preserves this file so you can edit it.
public sealed class PretzelEncryptionExample : MonoBehaviour
{
    [Serializable]
    public sealed class Profile
    {
        public string playerName = "Player";
        public int level = 1;
    }

    private const string ExampleFolder = "PretzelEncryptionExample";

    [ContextMenu("Encryption/JSON/AES - Save profile")]
    public void SaveAesWithJsonKeys()
    {
        CreateJsonStore(EncryptionMode.Aes).Save("profile", new Profile { level = 7 });
        Debug.Log("Saved with the AES key from GameEncryptionKeys.json.");
    }

    [ContextMenu("Encryption/JSON/AES - Load profile")]
    public void LoadAesWithJsonKeys()
    {
        Show(CreateJsonStore(EncryptionMode.Aes).Load<Profile>("profile"));
    }

    [ContextMenu("Encryption/JSON/RSA - Write report")]
    public void WriteRsaWithJsonKeys()
    {
        CreateJsonStore(EncryptionMode.Rsa).Save("report", new Profile { level = 12 });
        Debug.Log("RSA report saved under persistentDataPath/" + ExampleFolder + "/Json. Send it to your server.");
    }

    [ContextMenu("Encryption/Embedded script/AES - Save profile")]
    public void SaveAesWithEmbeddedKeys()
    {
        CreateEmbeddedStore(EncryptionMode.Aes).Save("profile", new Profile { level = 7 });
        Debug.Log("Saved with the AES key supplied by the registered setup script.");
    }

    [ContextMenu("Encryption/Embedded script/AES - Load profile")]
    public void LoadAesWithEmbeddedKeys()
    {
        Show(CreateEmbeddedStore(EncryptionMode.Aes).Load<Profile>("profile"));
    }

    [ContextMenu("Encryption/Embedded script/RSA - Write report")]
    public void WriteRsaWithEmbeddedKeys()
    {
        CreateEmbeddedStore(EncryptionMode.Rsa).Save("report", new Profile { level = 12 });
        Debug.Log("RSA report saved under persistentDataPath/" + ExampleFolder + "/Embedded. Send it to your server.");
    }

    private static DataStore CreateJsonStore(EncryptionMode mode)
    {
        // Reads Assets/Resources/Pretzel/GameEncryptionKeys.json in the Editor AND the built game.
        // This deliberately bypasses providers, environment variables and Editor-only keys.
        EncryptionKeys keys = GameEncryptionKeyConfig.Load();
        return CreateStore(keys, mode, ExampleFolder + "/Json");
    }

    private static DataStore CreateEmbeddedStore(EncryptionMode mode)
    {
        // PretzelDataSetup registers this provider BeforeSceneLoad. Call from Start or later, in Play mode.
        // No reference to PretzelDataSetup is needed, so the example also compiles with JSON-only exports.
        // This uses the currently registered provider; a token/custom provider can replace the embedded one.
        if (EncryptionKeys.Provider == null)
        {
            throw new InvalidOperationException("No setup provider registered. Export embedded C# and restart Play mode.");
        }

        EncryptionKeys keys = EncryptionKeys.Provider.Invoke();
        return CreateStore(keys, mode, ExampleFolder + "/Embedded");
    }

    [ContextMenu("Encryption/AES - Save profile")]
    public void SaveAesProfile()
    {
        CreateStore(EncryptionMode.Aes).Save("profile", new Profile { level = 7 });
        Debug.Log("Encrypted example profile saved. Use Load profile to read it.");
    }

    [ContextMenu("Encryption/AES - Load profile")]
    public void LoadAesProfile()
    {
        Show(CreateStore(EncryptionMode.Aes).Load<Profile>("profile"));
    }

    [ContextMenu("Encryption/RSA - Write report for server")]
    public void WriteRsaReport()
    {
        CreateStore(EncryptionMode.Rsa).Save("report", new Profile { level = 12 });
        Debug.Log("Send report.pretzel from persistentDataPath/" + ExampleFolder
                  + " over HTTPS to your server. The public key cannot decrypt it.");
    }

    private static DataStore CreateStore(EncryptionMode mode)
    {
        DataIOConfig config = DataIOConfig.Load();
        if (!config || config.format != DataFormat.Encrypted || config.encryptionMode != mode)
        {
            throw new InvalidOperationException("Choose Encrypted + " + mode + " in Pretzel > Data Format first.");
        }

        // Same API for exported JSON, embedded PretzelDataSetup and environment variables.
        // Provider > environment > Editor-only file > bundled JSON. Call after startup setup/sign-in.
        EncryptionKeys keys = EncryptionKeys.Resolve(config.environmentPrefix);
        return CreateStore(keys, mode, ExampleFolder);
    }

    private static DataStore CreateStore(EncryptionKeys keys, EncryptionMode mode, string folder)
    {
        if (keys == null || (mode == EncryptionMode.Aes ? !keys.HasAesKey : !keys.HasPublicKey))
        {
            throw new InvalidOperationException("No key for this mode. Generate and export it on the Keys tab first.");
        }

        // An explicit store keeps the example out of your real save folder.
        // Strip any Editor-only private key so this client example only has the public key.
        EncryptionKeys clientKeys = mode == EncryptionMode.Aes
            ? new EncryptionKeys { AesKey = keys.AesKey }
            : new EncryptionKeys { RsaPublicKey = keys.RsaPublicKey };
        return DataStoreFactory.Create(DataRoots.PersistentData(folder),
            DataFormat.Encrypted, clientKeys, mode);
    }

    // Never generate a new key each launch or log keys/tokens. Back up keys needed by old saves.

    public static async Awaitable SaveAndLoadAfterSignInAsync(CancellationToken cancellationToken = default)
    {
        DataStore store = CreateStore(EncryptionMode.Aes);
        await store.SaveAsync("signed-in-profile", new Profile(), cancellationToken);
        Show(await store.LoadAsync<Profile>("signed-in-profile", cancellationToken));
    }

    // TRUSTED SERVER/TOOL ONLY: supply the private key from a secret store, never from a game export.
    public static LoadResult<Profile> ReadRsaReportOnServer(string receivedFolder, EncryptionKeys serverKeys)
    {
        if (serverKeys == null || !serverKeys.HasPrivateKey)
        {
            throw new InvalidOperationException("The server needs its RSA private key. Never ship it in the game.");
        }

        DataStore server = DataStoreFactory.Create(DataRoots.Absolute(receivedFolder),
            DataFormat.Encrypted, new EncryptionKeys { RsaPrivateKey = serverKeys.RsaPrivateKey }, EncryptionMode.Rsa);
        LoadResult<Profile> result = server.Load<Profile>("report");
        Show(result);
        return result;
    }

    private static void Show(LoadResult<Profile> result)
    {
        if (result.Ok)
        {
            Debug.Log($"Loaded {result.Value.playerName}, level {result.Value.level}. Backup: {result.RecoveredFromBackup}");
        }
        else if (result.Status == DataStatus.NotFound)
        {
            Debug.Log("No example save yet. Save a profile first.");
        }
        else
        {
            Debug.LogError($"Wrong key or damaged example save: {result.Error}. Do not overwrite it automatically.");
        }
    }
}
