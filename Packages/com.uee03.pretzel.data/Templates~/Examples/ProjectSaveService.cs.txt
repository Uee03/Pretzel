using System;
using System.Collections.Generic;
using System.IO;
using System.Threading;
using Pretzel.Data;
using UnityEngine;

/// <summary>One project-owned API for profile slots. Gameplay does not construct stores or choose formats.</summary>
public sealed class ProjectSaveService
{
    private readonly IDataStore _store;

    public ProjectSaveService(IDataStore store)
    {
        _store = store ?? throw new ArgumentNullException(nameof(store));
    }

    public async Awaitable<ProjectSaveData> LoadOrCreateAsync(int slot, CancellationToken cancellationToken = default)
    {
        LoadResult<ProjectSaveData> result = await _store.LoadAsync<ProjectSaveData>(ProfileKey(slot), cancellationToken);
        switch (result.Status)
        {
            case DataStatus.Ok:
                if (result.Value == null)
                {
                    throw new InvalidDataException("The profile contains null data. Preserve the file and offer recovery.");
                }

                if (result.RecoveredFromBackup)
                {
                    Debug.LogWarning("[Project saves] The main profile was corrupt; loaded its backup.");
                }

                return result.Value;
            case DataStatus.NotFound:
                // A new profile stays in memory until the player explicitly saves.
                return new ProjectSaveData();
            default:
                throw new InvalidDataException($"Cannot load slot {slot}: {result.Error}. "
                    + "Do not overwrite it with a new profile. Offer retry, backup recovery or an explicit reset.");
        }
    }

    public Awaitable SaveAsync(int slot, ProjectSaveData profile, CancellationToken cancellationToken = default)
    {
        if (profile == null)
        {
            throw new ArgumentNullException(nameof(profile));
        }

        // Snapshot before background serialization; gameplay can keep changing the live profile.
        ProjectSaveData snapshot = new()
        {
            playerName = profile.playerName,
            level = profile.level,
            coins = profile.coins
        };
        return _store.SaveAsync(ProfileKey(slot), snapshot, cancellationToken);
    }

    public async Awaitable<IReadOnlyList<int>> ListSlotsAsync(CancellationToken cancellationToken = default)
    {
        IReadOnlyList<int> folders = await _store.ListSlotsAsync("Profiles", cancellationToken);
        List<int> profiles = new();
        foreach (int slot in folders)
        {
            // Slot enumeration includes folders left behind after their files were deleted.
            if (await _store.ExistsAsync(ProfileKey(slot), cancellationToken))
            {
                profiles.Add(slot);
            }
        }

        return profiles;
    }

    /// <summary>Call only after the player confirms that the slot should be permanently deleted.</summary>
    public Awaitable<bool> DeleteSlotAsync(int slot, CancellationToken cancellationToken = default)
    {
        return _store.DeleteAsync(ProfileKey(slot), cancellationToken);
    }

    private static DataKey ProfileKey(int slot)
    {
        return new DataKey("profile", slot: slot, folder: "Profiles");
    }
}
