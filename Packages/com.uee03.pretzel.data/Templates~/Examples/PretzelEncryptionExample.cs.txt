using System;
using System.Text;
using Pretzel.Data;
using UnityEngine;

// No keys are stored here. Export them on the Keys tab, then use the Inspector buttons.
public sealed class PretzelEncryptionExample : MonoBehaviour
{
    public enum KeyOrigin { Json, EmbeddedScript }

    public KeyOrigin source = KeyOrigin.Json;
    [HideInInspector] public EncryptionMode mode = EncryptionMode.Aes;
    [TextArea(3, 10)] public string plainText = "Hello from Pretzel!";
    [TextArea(3, 10)] public string encryptedText = "";

    public void Encrypt()
    {
        EncryptionTransform encryption = new(mode, GetKeys());
        encryptedText = Convert.ToBase64String(encryption.Encode(Encoding.UTF8.GetBytes(plainText ?? "")));
    }

    public void Decrypt()
    {
        if (string.IsNullOrWhiteSpace(encryptedText))
        {
            throw new InvalidOperationException("Paste or generate encrypted text first.");
        }

        EncryptionTransform encryption = new(mode, GetKeys());
        byte[] decrypted = encryption.Decode(Convert.FromBase64String(encryptedText));
        plainText = Encoding.UTF8.GetString(decrypted);
    }

    public void DisplayPlainText() => Debug.Log(plainText, this);
    public void DisplayEncryptedText() => Debug.Log(encryptedText, this);

    private EncryptionKeys GetKeys()
    {
        EncryptionKeys keys;
        switch (source)
        {
            case KeyOrigin.Json:
                keys = GameEncryptionKeyConfig.Load();
                break;
            case KeyOrigin.EmbeddedScript:
                if (!Application.isPlaying || EncryptionKeys.Provider == null)
                {
                    throw new InvalidOperationException("Export embedded C# and enter Play mode so the setup registers its provider.");
                }

                keys = EncryptionKeys.Provider.Invoke();
                break;
            default:
                throw new ArgumentOutOfRangeException(nameof(source));
        }

        if (keys == null || !keys.HasAesKey)
        {
            throw new InvalidOperationException("No AES key in the selected export. Generate and export one on the Keys tab.");
        }

        return keys.ForClient();
    }
}
